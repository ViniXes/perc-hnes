rules_version = '2';
// =============================================================================
// Reglas de seguridad de Firestore para PERC-HNES.
// -----------------------------------------------------------------------------
// Pegalas en: Firebase Console > Firestore Database > Reglas > Publicar.
//
// CAMBIOS DE ESTA VERSION
//  1) Cuentas MINSAL: solo pueden leer lo que alimenta el monitoreo nacional de
//     hospitales (serviceTabulators, calendario y overrides) y su propio perfil.
//     Todo lo demas del HNES (SEPS, Horas, POA, Censo, Insumos, matrices) les
//     queda cerrado desde el servidor, no solo escondido en la pantalla.
//  2) Nadie puede ascenderse solo: un usuario ya no puede tocar su propio rol,
//     permisos, servicio ni la marca isMinsal de su documento. Antes podia.
//  3) Comite de Expediente Clinico: coleccion cecTabulators. La LEEN los miembros
//     del comite completa (las 13 listas), mas Direccion, supervisores y los jefes
//     de division. La ESCRIBEN solo los administradores y el miembro del comite AL
//     QUE SE LE ASIGNO ESA LISTA: el campo 'cecServicios' de su perfil dice cuales
//     puede llenar; si intenta guardar otra, el servidor se lo rechaza aunque la
//     pantalla se lo permitiera. 'cec' y 'cecServicios' son campos que solo un
//     administrador puede escribir, para que nadie se agregue solo al comite ni se
//     asigne listas ajenas.
//  4) Gestor del comite: si el perfil tiene cec == true y cecGestor == true, puede
//     desbloquear y bloquear las listas del comite fuera de fecha (documento
//     documentControl/cecDesbloqueos). 'cecGestor' solo lo escribe un admin.
//  5) Bitacora (auditLog): toda cuenta con sesion deja constancia de sus acciones
//     sensibles (desbloqueos, cierres, permisos, claves, bajas). Solo el
//     administrador la LEE y nadie la edita ni la borra: es un registro.
//  6) Acceso "Monitoreo del C.E. Clinico" (menuGrants incluye 'panel-monitor-cec'):
//     la cuenta puede LEER las listas del comite (cecTabulators) para ver su
//     avance en "Monitoreo general". No puede escribirlas. 'menuGrants' solo lo
//     escribe un admin, asi que nadie se lo puede dar a si mismo.
//  7) Cierre de PERC en el servidor: captureWindows/{mes}.percCierre guarda la
//     hora de cierre (2:30 p. m. del 3er dia habil). Despues de esa hora nadie
//     que no sea administrador puede guardar el PERC de ese mes, salvo que el
//     tablero tenga una reapertura vigente (captureOverrides, 24 horas). Si un
//     supervisor lo cerro a mano, tampoco se puede guardar.
//  8) POA (Plan Anual Operativo): nadie entra por ser de un servicio. Lo leen y
//     escriben los administradores (todos los PAO) y la cuenta con el acceso
//     'panel-poa', SOLO el PAO de su unidad: la que le asigno el administrador
//     ('poaUnidad': un servicio, una Division o la Subdireccion Administrativa) o,
//     si no tiene, su propio servicio. 'poaUnidad' solo lo escribe un admin.
// =============================================================================
service cloud.firestore {
  match /databases/{database}/documents {
    function isSignedIn() {
      return request.auth != null;
    }
    function isAdmin() {
      return isSignedIn()
        && (request.auth.token.email == 'hcardoza.admin@perc-hnes.app'
            || request.auth.token.email == 'ffuentes@perc-hnes.app'
            || request.auth.token.email == 'amontes@perc-hnes.app'
            || request.auth.token.email == 'jcmiranda@perc-hnes.app'
            || request.auth.token.email == 'dalvarez@perc-hnes.app'
            || request.auth.token.email == 'jhernandez@perc-hnes.app'
            || request.auth.token.email == 'vhernandez@perc-hnes.app'
            || request.auth.token.email == 'rcenteno@perc-hnes.app');
    }
    function userProfile() {
      return get(/databases/$(database)/documents/serviceUsers/$(request.auth.uid)).data;
    }
    function isSupervisor() {
      return isSignedIn() && userProfile().role == 'supervisor';
    }

    // --- MINSAL ---------------------------------------------------------------
    // Cuenta del nivel ministerial: entra solo al monitoreo nacional de
    // hospitales. Se comprueba con exists() antes de leer para que una cuenta
    // recien creada, o sin documento de perfil, no rompa la evaluacion.
    function tienePerfil() {
      return isSignedIn()
        && exists(/databases/$(database)/documents/serviceUsers/$(request.auth.uid));
    }
    function esMinsal() {
      return tienePerfil() && userProfile().get('isMinsal', false) == true;
    }
    // Se usa en todo lo que NO le corresponde ver ni tocar a una cuenta MINSAL.
    function noEsMinsal() {
      return !esMinsal();
    }

    // --- Comite de Expediente Clinico ----------------------------------------
    function esComite() {
      return tienePerfil() && userProfile().get('cec', false) == true;
    }
    // Jefe de division: tiene division asignada, o una division que monitorea.
    function esJefeDivision() {
      return tienePerfil()
        && (userProfile().get('division', '') != ''
            || userProfile().get('monitorDivision', '') != '');
    }
    function esDireccion() {
      return tienePerfil() && userProfile().get('isDirector', false) == true;
    }
    // El comite VE las 13 listas, pero cada quien LLENA solo la de su servicio.
    // 'cecServicios' es la lista de ids que el administrador le asigno. Si esta
    // vacia, la cuenta entra en solo lectura.
    function llenaListaDelComite() {
      return esComite()
        && userProfile().get('cecServicios', [])
             .hasAny([request.resource.data.serviceId]);
    }

    // Campos del perfil que SOLO un administrador puede escribir. Son los que
    // definen quien es cada quien: si un usuario pudiera tocarlos, se daria a si
    // mismo el rol que quisiera. Lo demas de su documento (contrasena cambiada,
    // aceptacion de politicas, documento de identidad) si lo escribe el mismo.
    function camposDePrivilegio() {
      return [
        'role', 'permissions', 'isDirector', 'isMinsal', 'isActive', 'isChief',
        'serviceId', 'serviceName', 'menuGrants', 'noCapture', 'supervisorModules',
        'captureModules', 'viewPerc', 'viewSeps', 'viewHoras', 'division',
        'department', 'monitorDivision', 'username', 'loginEmail', 'sepsTables',
        'cec', 'cecServicios', 'cecGestor', 'poaUnidad'
      ];
    }
    function noTocaPrivilegios() {
      return !request.resource.data.diff(resource.data)
        .affectedKeys().hasAny(camposDePrivilegio());
    }

    match /serviceUsers/{uid} {
      allow read: if isAdmin() || (isSignedIn() && request.auth.uid == uid);
      // Crear cuentas es cosa del administrador (es quien aprueba las solicitudes).
      allow create: if isAdmin();
      // El propio usuario actualiza su documento, pero sin tocar su privilegio.
      allow update: if isAdmin()
        || (isSignedIn() && request.auth.uid == uid && noTocaPrivilegios());
      allow delete: if isAdmin();
    }
    match /serviceAssignments/{serviceId} {
      allow read, write: if isAdmin();
    }
    // --- Cierre de PERC ------------------------------------------------------
    // Si el mes todavia no tiene hora de cierre publicada, no se bloquea nada
    // (la publica el administrador al entrar a PULSO).
    function percEnPlazo(periodId) {
      let ref = /databases/$(database)/documents/captureWindows/$(periodId);
      return !exists(ref) || request.time < get(ref).data.percCierre;
    }
    function percOverride(periodId, serviceId) {
      let id = periodId + '__' + serviceId + '__perc';
      return /databases/$(database)/documents/captureOverrides/$(id);
    }
    // Reapertura vigente: estado 'open' de menos de 24 horas.
    function percReabierto(periodId, serviceId) {
      let ref = percOverride(periodId, serviceId);
      return exists(ref)
        && get(ref).data.state == 'open'
        && request.time < get(ref).data.updatedAt + duration.value(24, 'h');
    }
    function percCerradoAMano(periodId, serviceId) {
      let ref = percOverride(periodId, serviceId);
      return exists(ref) && get(ref).data.state == 'closed';
    }
    function percAbierto(periodId, serviceId) {
      return percReabierto(periodId, serviceId)
        || (percEnPlazo(periodId) && !percCerradoAMano(periodId, serviceId));
    }
    match /captureWindows/{periodId} {
      allow read: if isSignedIn();
      allow write: if isAdmin() || (tienePerfil() && userProfile().role == 'admin');
    }
    // PRODUCCION DISTRIBUIDA (PERC). Es lo unico del HNES que una cuenta MINSAL
    // puede LEER: de aca sale el avance del Hospital Nacional El Salvador dentro
    // del monitoreo de hospitales. Escribir, nunca.
    match /serviceTabulators/{docId} {
      allow read: if isSignedIn();
      allow create, update: if isAdmin()
        || (tienePerfil() && userProfile().role == 'admin')
        || (isSignedIn() && noEsMinsal()
            && request.resource.data.userId == request.auth.uid
            && percAbierto(request.resource.data.periodId, request.resource.data.serviceId));
      allow delete: if isAdmin();
    }
    match /sepsTabulators/{docId} {
      allow read: if isSignedIn() && noEsMinsal();
      allow create, update: if isAdmin()
        || (isSignedIn() && noEsMinsal()
            && request.resource.data.userId == request.auth.uid)
        || request.auth.token.email == 'jcmiranda@perc-hnes.app';
      allow delete: if isAdmin();
    }
    match /horasTabulators/{docId} {
      allow read: if isSignedIn() && noEsMinsal();
      allow create, update: if isAdmin()
        || (isSignedIn() && noEsMinsal()
            && request.resource.data.userId == request.auth.uid);
      allow delete: if isAdmin();
    }
    // El calendario si lo lee MINSAL: el tablero lo necesita para saber de que
    // mes esta hablando. No trae ninguna cifra.
    match /captureCalendar/{periodId} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
    }
    match /captureOverrides/{docId} {
      allow read: if isSignedIn();
      allow create, update: if isAdmin()
        || (isSupervisor()
            && userProfile().supervisorModules.hasAny([request.resource.data.moduleId]));
      allow delete: if isAdmin()
        || (isSupervisor()
            && userProfile().supervisorModules.hasAny([resource.data.moduleId]));
    }
    match /captureRequests/{docId} {
      allow read: if isSignedIn() && noEsMinsal();
      allow create: if isSignedIn() && noEsMinsal()
        && request.resource.data.requestedByUid == request.auth.uid;
      allow update: if isAdmin()
        || (isSupervisor()
            && userProfile().supervisorModules.hasAny([resource.data.moduleId]))
        || (isSignedIn() && resource.data.requestedByUid == request.auth.uid);
      allow delete: if isAdmin();
    }
    // BITACORA: registro de acciones sensibles. Cualquier cuenta con sesion la
    // ESCRIBE (queda constancia de lo que hace, siempre con su propia identidad),
    // pero solo el admin la LEE, y nadie puede editarla ni borrarla.
    match /auditLog/{docId} {
      allow read: if isAdmin();
      allow create: if isSignedIn() && request.resource.data.userId == request.auth.uid;
      allow update, delete: if false;
    }
    // Desbloqueos del Comite de Expediente Clinico: los escribe el admin o el
    // gestor del comite (cec == true y cecGestor == true).
    match /documentControl/cecDesbloqueos {
      allow read: if isSignedIn() && noEsMinsal();
      allow write: if isAdmin()
        || (esComite() && userProfile().get('cecGestor', false) == true);
    }
    match /documentControl/{year} {
      allow read: if isSignedIn() && noEsMinsal();
      allow write: if isAdmin();
    }
    // Estructura (matriz) de los tabuladores SEPS: un documento por servicio con
    // las tablas y filas agregadas, ocultas o renombradas. La LEE cualquier usuario
    // autenticado que no sea MINSAL; la ESCRIBEN los administradores y el revisor.
    match /sepsLayouts/{serviceId} {
      allow read: if isSignedIn() && noEsMinsal();
      allow write: if isAdmin()
        || request.auth.token.email == 'jcmiranda@perc-hnes.app'
        || (isSignedIn() && userProfile().role == 'admin');
    }
    // POA - Plan Anual Operativo por unidad y año ("<unidad>__<año>" y
    // "<unidad>__<año>__media"); la unidad es un servicio, una División o la
    // Subdirección Administrativa. Nadie entra por ser de un servicio: lo ven y
    // editan los administradores (todos) y la cuenta a la que un administrador le
    // otorgó el acceso "POA" (menuGrants 'panel-poa'), SOLO el de su unidad.
    function adminPoa() {
      return isAdmin() || (tienePerfil() && userProfile().role == 'admin');
    }
    // Unidad del PAO de la cuenta: la que le asignó el administrador (poaUnidad,
    // p. ej. "division-medica") o, si no tiene, su propio servicio.
    function unidadPoa() {
      let p = userProfile();
      let u = p.get('poaUnidad', '');
      let s = p.get('serviceId', '');
      return (u is string && u != '') ? u : ((s is string) ? s : '');
    }
    function tieneAccesoPoa() {
      return tienePerfil() && noEsMinsal()
        && userProfile().get('menuGrants', []).hasAny(['panel-poa'])
        && unidadPoa() != '';
    }
    function poaDeSuUnidad(docId) {
      return tieneAccesoPoa() && docId.matches(unidadPoa() + '__.*');
    }
    match /poaDocuments/{docId} {
      allow get: if adminPoa() || poaDeSuUnidad(docId);
      allow list: if adminPoa()
        || (tieneAccesoPoa() && resource.data.serviceId == unidadPoa());
      allow create, update: if adminPoa()
        || (poaDeSuUnidad(docId) && request.resource.data.serviceId == unidadPoa());
      allow delete: if adminPoa();
    }
    // Censo Diario de Pacientes (id = "<periodo>", p.ej. "2026-06").
    match /censoDiario/{periodId} {
      allow read: if isSignedIn() && noEsMinsal()
        && (isAdmin()
            || userProfile().role == 'admin'
            || userProfile().role == 'supervisor');
      allow write: if isSignedIn()
        && (isAdmin()
            || userProfile().role == 'admin'
            || request.auth.token.email == 'amontes@perc-hnes.app');
    }
    // Insumos de Almacen (id = "<periodo>", p.ej. "2026-06"). Guardado por mes.
    match /insumosAlmacen/{periodId} {
      allow read: if isSignedIn() && noEsMinsal()
        && (isAdmin()
            || userProfile().role == 'admin'
            || userProfile().role == 'supervisor'
            || userProfile().serviceId == 'almacen');
      allow write: if isSignedIn()
        && (isAdmin()
            || userProfile().role == 'admin'
            || userProfile().serviceId == 'almacen');
    }
    // COMITE DE EXPEDIENTE CLINICO. Un documento por mes y por servicio
    // ("<periodo>__<servicio>"): las listas de monitoreo del comite.
    match /cecTabulators/{docId} {
      allow read: if isAdmin()
        || esComite()
        || esDireccion()
        || esJefeDivision()
        || isSupervisor()
        // Acceso otorgado "Monitoreo del C.E. Clinico" (solo ver el avance).
        || (tienePerfil()
            && userProfile().get('menuGrants', []).hasAny(['panel-monitor-cec']));
      // Escribe el administrador, o el miembro del comite al que se le asigno
      // ESTA lista. Guardar la de otro servicio se rechaza desde el servidor.
      allow create, update: if isAdmin()
        || (llenaListaDelComite() && request.resource.data.userId == request.auth.uid);
      allow delete: if isAdmin();
    }
    // Solicitudes de registro (jefes de servicio, direccion, MINSAL y comite).
    match /signupRequests/{id} {
      allow read: if isAdmin();
      allow create: if request.resource.data.status == 'pending'
        && request.resource.data.firstName is string
        && request.resource.data.lastName is string
        && request.resource.data.email is string
        && request.resource.data.serviceId is string
        && request.resource.data.serviceName is string;
      allow update, delete: if isAdmin();
    }
    // Centro de Soporte. Una cuenta MINSAL SI puede abrir un ticket (tiene el
    // boton de Soporte en su menu), pero no puede leer los de nadie.
    match /supportTickets/{ticketId} {
      allow read: if isAdmin() || isSupervisor();
      allow create: if isSignedIn()
        && request.resource.data.status == 'pendiente'
        && request.resource.data.reporterUid == request.auth.uid
        && request.resource.data.message is string
        && request.resource.data.message.size() > 0;
      allow update: if isAdmin() || isSupervisor();
      allow delete: if isAdmin();
    }
    // Registro global de documentos de identidad (DUI/pasaporte/carne).
    match /documentRegistry/{docId} {
      allow get: if true;
      allow list: if isAdmin();
      allow create: if true;
      allow update, delete: if isAdmin();
    }
  }
}
